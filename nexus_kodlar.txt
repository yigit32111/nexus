// Simple in-memory per-user rate limiter (25 requests / minute)
const rateMap = new Map<string, { count: number; resetAt: number }>();
function checkRateLimit(userId: string): boolean {
  const now = Date.now();
  const entry = rateMap.get(userId);
  if (!entry || now > entry.resetAt) {
    rateMap.set(userId, { count: 1, resetAt: now + 60_000 });
    return true;
  }
  if (entry.count >= 25) return false;
  entry.count += 1;
  return true;
}

function fail(message: string, status = 400) {

function validImage(type: string, buffer: Buffer) {
  if (type === "image/jpeg")
    return buffer.length >= 3 && buffer[0] === 0xff && buffer[1] === 0xd8 && buffer[2] === 0xff;
  if (type === "image/png")
    return (
      buffer.length >= 8 &&
      buffer.subarray(0, 8).equals(Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]))
    );
  if (type === "image/webp")
    return (
      buffer.length >= 12 &&
      buffer.toString("ascii", 0, 4) === "RIFF" &&
      buffer.toString("ascii", 8, 12) === "WEBP"
    );
  return false;
}

import { NextResponse, type NextRequest } from "next/server";
import { createClient } from "../../../utils/supabase/server";

export async function GET(request: NextRequest) {
  const url = new URL(request.url);
  const code = url.searchParams.get("code");
  const next = url.searchParams.get("next");
  const destination = next && next.startsWith("/") && !next.startsWith("//") ? next : "/";

  if (code) {
    const supabase = await createClient();
    const { error } = await supabase.auth.exchangeCodeForSession(code);
    if (!error) return NextResponse.redirect(new URL(destination, url.origin));
  }

  return NextResponse.redirect(new URL("/login?auth=callback_error", url.origin));
}

  async function send(text?: string, opts?: { regenerateFrom?: Message[]; editIndex?: number }) {

    let outgoing: Message[];
    if (opts?.regenerateFrom) outgoing = baseMessages;
    else if (opts?.editIndex !== undefined) outgoing = [...baseMessages.slice(0, opts.editIndex), { id: uid(), role: "user", content, createdAt: Date.now(), attachments: meta }];
    else outgoing = [...baseMessages, { id: uid(), role: "user", content: content || "EklediÄŸim dosyalarÄ± incele ve aÃ§Ä±kla.", createdAt: Date.now(), attachments: meta }];
    const userText = content || "EklediÄŸim dosyalarÄ± incele ve aÃ§Ä±kla.";

      return;

      const toolsModel = prefs.model || "openai/gpt-oss-20b";
      if (!/^openai\/gpt-oss-(20b|120b)$/.test(toolsModel)) {
        setNotice("Groq Web Arama ve Kod Ã‡alÄ±ÅŸtÄ±rma araÃ§larÄ± openai/gpt-oss-20b veya openai/gpt-oss-120b modellerini gerektirir. Modeli ayarlardan deÄŸiÅŸtir.");
        return;
      }
    }

    if (!opts?.regenerateFrom) detectMemorySuggestion(userText);
    patchMessages(targetChatId, outgoing);
    setInput(""); setFiles([]); setEditingMessageId(null); setBusy(true); setNotice("");

    const controller = new AbortController();

      const res = await fetch("/api/chat", {

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";

export async function updateSession(request: NextRequest) {
  let response = NextResponse.next({ request });

  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const supabaseKey =
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY ||
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;

  if (!supabaseUrl || !supabaseKey) {
    return response;
  }

  const supabase = createServerClient(supabaseUrl, supabaseKey, {
    cookies: {
      getAll() {
        return request.cookies.getAll();
      },
      setAll(cookiesToSet) {
        cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value));
        response = NextResponse.next({ request });
        cookiesToSet.forEach(({ name, value, options }) =>
          response.cookies.set(name, value, options)
        );
      },
    },
  });

  const {
    data: { user },
  } = await supabase.auth.getUser();

  const pathname = request.nextUrl.pathname;
  const isPublicPage =
    pathname === "/login" ||
    pathname === "/reset-password" ||
    pathname.startsWith("/auth/callback") ||
    pathname.startsWith("/_next") ||
    pathname === "/favicon.ico";
  const isApiRoute = pathname.startsWith("/api/");

  if (!user && !isPublicPage && !isApiRoute) {
    const url = request.nextUrl.clone();
    url.pathname = "/login";
    url.searchParams.set("next", pathname);
    const redirect = NextResponse.redirect(url);
    response.cookies.getAll().forEach((cookie) => redirect.cookies.set(cookie));
    return redirect;
  }

  if (user && pathname === "/login") {
    const url = request.nextUrl.clone();
    url.pathname = "/";
    url.search = "";
    const redirect = NextResponse.redirect(url);
    response.cookies.getAll().forEach((cookie) => redirect.cookies.set(cookie));
    return redirect;
  }

  return response;
}
